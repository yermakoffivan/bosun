# Subscriptions from a control client attached to __bosun_monitor; panes %0..%5 live in OTHER sessions.
ctl refresh-client -B 'other_pane:%0:#{session_name}|#{pane_title}|#{pane_current_command}'
ctl refresh-client -B 'other_pane6:%5:#{session_name}|#{pane_title}|#{pane_current_command}'
ctl refresh-client -B 'own_all:%*:#{session_name}|#{pane_title}|#{pane_current_command}'
ctl refresh-client -B 'other_window:@1:#{session_name}|#{window_name}'

sleep 2.5
